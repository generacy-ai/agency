# Data Model: Git Utilities

## Core Types

### GitStatus

The minimal status interface for git repository state:

```typescript
/**
 * Minimal git status information
 */
export interface GitStatus {
  /** Whether the working tree has no uncommitted changes */
  isClean: boolean;

  /** Current branch name (or "HEAD" if detached) */
  currentBranch: string;

  /** Whether there are staged or unstaged changes */
  hasChanges: boolean;
}
```

### FetchOptions

Options for the fetch operation:

```typescript
/**
 * Options for git fetch operation
 */
export interface FetchOptions {
  /** Fetch from all remotes (default: true) */
  all?: boolean;

  /** Prune deleted remote branches (default: true) */
  prune?: boolean;
}
```

### Branch Operations

```typescript
/**
 * Get current branch name
 * @param cwd - Working directory (defaults to process.cwd())
 * @returns Branch name (or "HEAD" if detached)
 * @throws Error if not a git repository
 */
export async function getCurrentBranch(cwd?: string): Promise<string>;

/**
 * Check if branch exists locally or on origin
 * @param name - Branch name to check
 * @param cwd - Working directory (defaults to process.cwd())
 * @returns true if branch exists locally or as origin/<name>
 * @throws Error if not a git repository
 */
export async function branchExists(name: string, cwd?: string): Promise<boolean>;

/**
 * Create a new branch and optionally checkout
 * @param name - New branch name
 * @param from - Base branch/ref (defaults to current HEAD)
 * @param cwd - Working directory (defaults to process.cwd())
 * @throws Error if not a git repository or branch already exists
 */
export async function createBranch(
  name: string,
  from?: string,
  cwd?: string
): Promise<void>;

/**
 * Switch to a different branch
 * @param branch - Branch name to checkout
 * @param cwd - Working directory (defaults to process.cwd())
 * @throws Error if not a git repository or branch doesn't exist
 */
export async function checkout(branch: string, cwd?: string): Promise<void>;
```
