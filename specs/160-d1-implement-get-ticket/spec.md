## Tool Definition

```typescript
export function createGetTicketTool(
  config: SpecKitConfig,
  getProvider: () => BacklogProvider,
): AgencyTool {
  return {
    name: 'spec_kit.get_ticket',
    description: 'Get ticket details from the configured backlog system',
    namespace: 'spec_kit',
    outputPattern: 'terse',
    modes: ['coding', 'research'],
    inputSchema: {
      type: 'object',
      properties: {
        ref: {
          type: 'string',
          description: 'Ticket reference - URL (https://github.com/owner/repo/issues/123) or identifier (#123, PROJ-123)',
        },
      },
      required: ['ref'],
    },
    async execute(params) {
      const { ref } = params as { ref: string };
      
      // 1. Auto-detect provider from ref if URL
      const ticketRef = detectTicketRef(ref, config.backlog.provider);
      if (!ticketRef) {
        return { content: [{ type: 'text', text: 'Invalid ticket reference' }], isError: true };
      }
      
      // 2. Get appropriate provider
      const provider = getProvider();
      
      // 3. Fetch ticket
      const ticket = await provider.getTicket(ticketRef.id);
      
      return {
        content: [{ type: 'text', text: JSON.stringify(ticket, null, 2) }],
      };
    },
  };
}
```
