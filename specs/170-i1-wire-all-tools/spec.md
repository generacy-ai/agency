// src/plugin.ts
export class SpecKitPlugin implements AgencyPlugin {
  readonly manifest = specKitManifest;
  
  private config!: SpecKitConfig;
  private provider!: BacklogProvider;
  private core!: AgencyCoreAPI;

  async initialize(core: AgencyCoreAPI): Promise<void> {
    this.core = core;
    
    // Load configuration
    const userConfig = core.getConfig<Partial<SpecKitConfig>>('plugins.spec-kit');
    this.config = resolveConfig(userConfig);
    
    // Initialize backlog provider
    this.provider = createProvider(this.config.backlog);
    
    // Create and register all tools
    const tools = createAllTools(this.config, () => this.provider, core);
    for (const tool of tools) {
      core.registerTool(tool);
    }
  }

  async shutdown(): Promise<void> {
    // Unregister tools
    for (const toolName of this.manifest.tools ?? []) {
      this.core.unregisterTool(toolName);
    }
  }
}
