  async getTicket(ref: string): Promise<Ticket> {
    const store = await this.loadStore();
    const ticket = store.tickets[ref];
    if (!ticket) throw new TicketNotFoundError(ref);
    return this.toTicket(ticket);
  }
  
  async createTicket(params: TicketCreateParams): Promise<Ticket> {
    const store = await this.loadStore();
    const id = `LOCAL-${String(store.nextId).padStart(3, '0')}`;
    store.tickets[id] = { ...params, id, state: 'open', createdAt: new Date() };
    store.nextId++;
    await this.saveStore(store);
    return this.toTicket(store.tickets[id]);
  }
