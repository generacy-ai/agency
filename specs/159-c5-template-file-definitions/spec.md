export const TEMPLATE_TYPES = ['spec', 'plan', 'tasks', 'checklist', 'agent-file'] as const;
export type TemplateType = typeof TEMPLATE_TYPES[number];

export interface TemplateDefinition {

  defaultFilename: string;

## Template Resolution

```typescript
export async function resolveTemplate(
  type: TemplateType,
  config: SpecKitConfig,
): Promise<string> {
  // 1. Check config.paths.templates for custom template
  const customPath = path.join(config.paths.templates, TEMPLATES[type].defaultFilename);
  if (await exists(customPath)) {
    return readFile(customPath);
  }
  
  // 2. Fall back to built-in default
  return TEMPLATES[type].defaultContent;
}
```
