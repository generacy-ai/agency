```typescript
test('get_ticket fetches Jira issue by key', async () => {
  const result = await callTool('spec_kit.get_ticket', { ref: 'PROJ-123' });
  expect(result.title).toBeDefined();
  expect(result.url).toMatch(/atlassian\.net/);
});
```

### Scenario 2: URL Parsing
```typescript
test('get_ticket parses Jira URLs', async () => {
  const result = await callTool('spec_kit.get_ticket', {
    ref: 'https://company.atlassian.net/browse/PROJ-123'
  });
  expect(result.ref.provider).toBe('jira');
  expect(result.ref.id).toBe('PROJ-123');
});
```
