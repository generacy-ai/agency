```typescript
export function detectTicketRef(
  input: string,
  defaultProvider: BacklogProvider['name'] = 'github'
): TicketRef | null {
  // 1. Try URL patterns first (most specific)
  if (isGitHubUrl(input)) return parseGitHubUrl(input);
  if (isJiraUrl(input)) return parseJiraUrl(input);
  if (isShortcutUrl(input)) return parseShortcutUrl(input);
  
  // 2. Try provider-specific patterns
  if (isJiraKey(input)) return { provider: 'jira', id: input, raw: input };
  if (isLocalKey(input)) return { provider: 'local', id: input, raw: input };
  
  // 3. GitHub-style patterns (#123 or plain number)
  if (isGitHubRef(input)) return { provider: 'github', id: input, raw: input };
  
  // 4. If numeric and no provider identified, use default
  if (/^\d+$/.test(input)) {
    return { provider: defaultProvider, id: input, raw: input };
  }
  
  return null;
}

// URL patterns
const GITHUB_URL = /github\.com\/[\w-]+\/[\w-]+\/issues\/(\d+)/;
const JIRA_URL = /\.atlassian\.net\/browse\/([\w]+-\d+)/;
const SHORTCUT_URL = /app\.shortcut\.com\/[\w-]+\/story\/(\d+)/;

// Shorthand patterns
const JIRA_KEY = /^[A-Z][A-Z0-9]+-\d+$/;
const LOCAL_KEY = /^LOCAL-\d+$/i;
const GITHUB_REF = /^#?\d+$/;
```
