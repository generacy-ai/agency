## Example Implementation

```typescript
// run.docker_compose_up
async function composeUp(params: {
  file?: string;
  services?: string[];
  detach?: boolean;
  build?: boolean;
}): Promise<ToolResult> {
  const args = ['compose'];
  if (params.file) args.push('-f', params.file);
  args.push('up');
  if (params.detach !== false) args.push('-d');
  if (params.build) args.push('--build');
  if (params.services) args.push(...params.services);
  
  const result = await exec('docker', args);
  return TerseOutput.fromExec(result, 'Services started.');
}
```
