### Scenario 1: Basic Ticket Fetch
```typescript
test('get_ticket fetches GitHub issue by number', async () => {
  const result = await callTool('spec_kit.get_ticket', { ref: '#1' });
  expect(result.title).toBeDefined();
  expect(result.url).toMatch(/github\.com/);
});
```

### Scenario 2: Create Feature from Issue
```typescript
test('create_feature from GitHub issue', async () => {
  const ticket = await callTool('spec_kit.get_ticket', { ref: '#1' });
  const feature = await callTool('spec_kit.create_feature', {
    description: ticket.title,
  });
  expect(feature.branch).toMatch(/^\d{3}-/);
  expect(feature.featureDir).toContain('specs/');
});
```

### Scenario 3: Tasks to Issues
```typescript
test('tasks_to_issues creates GitHub issues', async () => {
  const result = await callTool('spec_kit.tasks_to_issues', {
    dry_run: false,
    grouping: 'per-task',
  });
  expect(result.issues.every(i => i.status === 'created')).toBe(true);
});
```
