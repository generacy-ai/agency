```typescript
// src/utils/clarifications.ts

export interface ClarificationQuestion {
  number: number;
  topic: string;
  context: string;
  question: string;
  options?: Array<{ label: string; description: string }>;
  answer?: string;
  humancyRequestId?: string;  // For tracking Humancy response
}

export interface ClarificationBatch {
  timestamp: string;  // ISO date
  questions: ClarificationQuestion[];
}

export interface ClarificationsFile {
  batches: ClarificationBatch[];
}

export function parseClarifications(content: string): ClarificationsFile;
export function serializeClarifications(file: ClarificationsFile): string;
export function getNextQuestionNumber(file: ClarificationsFile): number;
export function findQuestion(file: ClarificationsFile, number: number): ClarificationQuestion | undefined;
export function updateAnswer(file: ClarificationsFile, number: number, answer: string): ClarificationsFile;
export function appendBatch(file: ClarificationsFile, questions: Omit<ClarificationQuestion, 'number'>[]): ClarificationsFile;
```
